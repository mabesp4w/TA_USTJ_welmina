class MemberController extends Controller
{
    public function index(Request $request): Response
    {
        $search = $request->query('search');
        $status = $request->query('status');

        $query = Member::with('memberships.package');

        if ($search) {
            $query->where(function ($q) use ($search) {
                $q->where('full_name', 'like', "%{$search}%")
                    ->orWhere('member_code', 'like', "%{$search}%")
                    ->orWhere('identity_no', 'like', "%{$search}%")
                    ->orWhere('phone', 'like', "%{$search}%");
            });
        }

        if ($status === 'Aktif') {
            $query->where('member_status', 'active');
        } elseif ($status === 'Nonaktif') {
            $query->where('member_status', 'inactive');
        }

        $members = $query
            ->latest()
            ->paginate(10)
            ->through(fn (Member $member) => [
                'id' => $member->id,
                'memberCode' => $member->member_code,
                'name' => $member->full_name,
                'plan' => $member->memberships->last()?->package?->package_name ?? '-',
                'period' => $member->memberships->last()
                    ? sprintf(
                        'Bergabung %s - Exp %s',
                        $member->memberships->last()->start_date->format('d M Y'),
                        $member->memberships->last()->end_date->format('d M Y'),
                    )
                    : '-',
                'status' => $member->member_status === 'active' ? 'Aktif' : 'Nonaktif',
            ]);

        $total = Member::count();
        $active = Member::where('member_status', 'active')->count();
        $expiring = MemberMembership::where('status', 'active')
            ->whereBetween('end_date', [now(), now()->addDays(7)])
            ->count();
        $inactive = Member::where('member_status', 'inactive')->count();

        return Inertia::render('admin/Member', [
            'stats' => [
                ['title' => 'Total Member', 'value' => (string) $total, 'subtitle' => 'Terdaftar di sistem'],
                ['title' => 'Aktif', 'value' => (string) $active, 'subtitle' => 'Masa berlaku aktif'],
                ['title' => 'Aktif Expired', 'value' => (string) $expiring, 'subtitle' => 'Dalam 7 hari ke depan'],
                ['title' => 'Nonaktif', 'value' => (string) $inactive, 'subtitle' => 'Masa berlaku habis'],
            ],
            'members' => $members,
        ]);
    }

    public function store(StoreMemberRequest $request): RedirectResponse
    {
        $birthDate = $request->date('birthDate');

        $member = Member::create([
            'member_code' => $this->generateMemberCode(),
            'full_name' => $request->input('fullName'),
            'identity_no' => $request->input('identityNo'),
            'phone' => $request->input('phone'),
            'birth_date' => $birthDate?->toDateString(),
            'gender' => $request->input('gender'),
            'email' => $request->input('email'),
            'address' => $request->input('address'),
            'member_status' => 'inactive',
            'joined_at' => now(),
            'created_by' => $request->user()?->id,
        ]);

        ActivityLogService::log(
            activityType: 'pendaftaran',
            activityName: 'Tambah Member: '.$member->full_name,
            description: 'Menambahkan member baru "'.$member->full_name.'" ('.$member->member_code.')',
            memberId: $member->id,
            referenceTable: 'members',
            referenceId: $member->id,
        );

        return redirect()->route('admin.member')->with('success', 'Data member berhasil disimpan!');
    }

    public function show(Member $member): JsonResponse
    {
        $member->load('memberships.package', 'creator');

        return response()->json([
            'member' => [
                'id' => $member->id,
                'memberCode' => $member->member_code,
                'fullName' => $member->full_name,
                'identityNo' => $member->identity_no,
                'phone' => $member->phone,
                'birthDate' => $member->birth_date?->format('Y-m-d'),
                'gender' => $member->gender,
                'email' => $member->email,
                'address' => $member->address,
                'memberStatus' => $member->member_status,
                'joinedAt' => $member->joined_at->format('d M Y'),
                'createdBy' => $member->creator?->full_name,
            ],
            'membership' => $member->memberships->last() ? [
                'packageName' => $member->memberships->last()->package?->package_name,
                'durationMonths' => $member->memberships->last()->duration_months,
                'startDate' => $member->memberships->last()->start_date->format('d M Y'),
                'endDate' => $member->memberships->last()->end_date->format('d M Y'),
                'status' => $member->memberships->last()->status,
            ] : null,
        ]);
    }

    public function update(UpdateMemberRequest $request, Member $member): RedirectResponse
    {
        $birthDate = $request->date('birthDate');

        $member->update([
            'full_name' => $request->input('fullName'),
            'identity_no' => $request->input('identityNo'),
            'phone' => $request->input('phone'),
            'birth_date' => $birthDate?->toDateString(),
            'gender' => $request->input('gender'),
            'email' => $request->input('email'),
            'address' => $request->input('address'),
            'member_status' => $request->input('memberStatus', $member->member_status),
        ]);

        ActivityLogService::log(
            activityType: 'pendaftaran',
            activityName: 'Update Member: '.$member->full_name,
            description: 'Memperbarui data member "'.$member->full_name.'" ('.$member->member_code.')',
            memberId: $member->id,
            referenceTable: 'members',
            referenceId: $member->id,
        );

        return redirect()->route('admin.member')->with('success', 'Data member berhasil diperbarui!');
    }

    public function destroy(Member $member): RedirectResponse
    {
        ActivityLogService::log(
            activityType: 'lainnya',
            activityName: 'Hapus Member: '.$member->full_name,
            description: 'Menghapus member "'.$member->full_name.'" ('.$member->member_code.')',
            memberId: $member->id,
            referenceTable: 'members',
            referenceId: $member->id,
        );

        $member->delete();

        return redirect()->route('admin.member')->with('success', 'Data member berhasil dihapus!');
    }

    protected function generateMemberCode(): string
    {
        $prefix = 'MB-';
        $last = Member::where('member_code', 'like', "$prefix%")
            ->orderBy('id', 'desc')
            ->value('member_code');

        if (! $last) {
            return $prefix.'0001';
        }

        $num = (int) substr($last, 3) + 1;

        return $prefix.str_pad((string) $num, 4, '0', STR_PAD_LEFT);
    }
}

function MemberPage({ stats, members }: { stats: Stat[]; members: PaginatedData<MemberItem> }) {
    const [statusFilter, setStatusFilter] = useState<string | null>(null);
    const [selectedMember, setSelectedMember] = useState<MemberItem | null>(null);
    const [editData, setEditData] = useState<EditMemberData | null>(null);
    const [editLoading, setEditLoading] = useState(false);
    const [detailData, setDetailData] = useState<DetailResponse | null>(null);
    const [detailLoading, setDetailLoading] = useState(false);
    const [deleteLoading, setDeleteLoading] = useState(false);

    const filteredMembers = statusFilter
        ? members.data.filter((m) => m.status === statusFilter)
        : members.data;

    const handleFilterChange = (filter: string | null) => {
        setStatusFilter(filter);
    };

    const openDetail = async (member: MemberItem) => {
        setSelectedMember(member);
        setDetailLoading(true);
        setDetailData(null);

        document.getElementById("detail-modal")?.showModal();

        try {
            const res = await fetch(show.url({ member: member.id }));
            const data: DetailResponse = await res.json();
            setDetailData(data);
        } catch {
            toast.error("Gagal memuat detail member");
        }

        setDetailLoading(false);
    };

    const openEdit = async (member: MemberItem) => {
        setEditData(null);
        setEditLoading(true);

        document.getElementById("edit-modal")?.showModal();

        try {
            const res = await fetch(show.url({ member: member.id }));
            const data: DetailResponse = await res.json();
            setEditData(data.member);
        } catch {
            toast.error("Gagal memuat data member");
        }

        setEditLoading(false);
    };

    const confirmDelete = (member: MemberItem) => {
        setSelectedMember(member);
        document.getElementById("delete-modal")?.showModal();
    };

    const handleDelete = () => {
        if (!selectedMember) return;

        setDeleteLoading(true);

        router.delete(destroy.url({ member: selectedMember.id }), {
            preserveScroll: true,
            onSuccess: () => {
                toast.success("Data member berhasil dihapus!");
                document.getElementById("delete-modal")?.close();
                setSelectedMember(null);
            },
            onError: () => {
                toast.error("Gagal menghapus data member");
            },
            onFinish: () => setDeleteLoading(false),
        });
    };

    const closeForm = () => {
        document.getElementById("member-modal")?.close();
    };

    const closeEditForm = () => {
        setEditData(null);
        document.getElementById("edit-modal")?.close();
    };

    // ... (JSX markup dibuang)
}

MemberPage.layout = (page: React.ReactNode) => <MemberLayout>{page}</MemberLayout>;

export default MemberPage;
