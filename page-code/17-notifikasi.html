Controller (PHP)

app/Http/Controllers/Admin/NotificationController.php 
class NotificationController extends Controller
{
    public function index(Request $request): Response
    {
        $type = $request->query('type');

        $query = Notification::with('creator');

        if ($type) {
            $query->where('notification_type', $type);
        }

        $notifications = $query
            ->orderByDesc('created_at')
            ->paginate(10)
            ->through(fn (Notification $n) => [
                'id' => $n->id,
                'title' => $n->title,
                'message' => $n->message,
                'notificationType' => $n->notification_type,
                'isRead' => $n->is_read,
                'createdAt' => $n->created_at?->format('Y-m-d H:i:s'),
            ]);

        $stats = [
            'total' => Notification::count(),
            'unread' => Notification::where('is_read', false)->count(),
            'recentToday' => Notification::whereDate('created_at', today())->count(),
        ];

        return Inertia::render('admin/Notification', [
            'notifications' => $notifications,
            'stats' => $stats,
        ]);
    }

    public function show(Notification $notification): JsonResponse
    {
        $notification->load('creator');

        if (! $notification->is_read) {
            $notification->update(['is_read' => true]);
        }

        return response()->json([
            'id' => $notification->id,
            'title' => $notification->title,
            'message' => $notification->message,
            'notificationType' => $notification->notification_type,
            'isRead' => true,
            'createdAt' => $notification->created_at?->format('Y-m-d H:i:s'),
            'relatedTable' => $notification->related_table,
            'relatedId' => $notification->related_id,
            'creatorName' => $notification->creator?->name,
        ]);
    }

    public function markAllAsRead(): RedirectResponse
    {
        Notification::where('is_read', false)->update(['is_read' => true]);

        return redirect()->route('admin.notification')->with('success', 'Semua notifikasi ditandai sudah dibaca!');
    }

    public function destroy(Notification $notification): RedirectResponse
    {
        $notification->delete();

        return redirect()->route('admin.notification')->with('success', 'Notifikasi berhasil dihapus!');
    }
}

function NotificationPage({
    notifications,
    stats,
}: {
    notifications: PaginatedData<NotificationItem>;
    stats: Stats;
}) {
    const [selectedNotification, setSelectedNotification] = useState<NotificationItem | null>(null);
    const [detailData, setDetailData] = useState<DetailResponse | null>(null);
    const [detailLoading, setDetailLoading] = useState(false);
    const [deleteLoading, setDeleteLoading] = useState(false);

    const openDetail = async (item: NotificationItem) => {
        setSelectedNotification(item);
        setDetailLoading(true);
        setDetailData(null);

        document.getElementById("detail-modal")?.showModal();

        try {
            const res = await fetch(show.url({ notification: item.id }));
            const data: DetailResponse = await res.json();
            setDetailData(data);
        } catch {
            toast.error("Gagal memuat detail notifikasi");
        }

        setDetailLoading(false);
    };

    const confirmDelete = (item: NotificationItem) => {
        setSelectedNotification(item);
        document.getElementById("delete-modal")?.showModal();
    };

    const handleDelete = () => {
        if (!selectedNotification) return;

        setDeleteLoading(true);

        router.delete(destroy.url({ notification: selectedNotification.id }), {
            preserveScroll: true,
            onSuccess: () => {
                toast.success("Notifikasi berhasil dihapus!");
                document.getElementById("delete-modal")?.close();
                setSelectedNotification(null);
            },
            onError: () => {
                toast.error("Gagal menghapus notifikasi");
            },
            onFinish: () => setDeleteLoading(false),
        });
    };

    const statCards = [
        { title: "Total Notifikasi", value: String(stats.total), subtitle: "Seluruh notifikasi" },
        { title: "Belum Dibaca", value: String(stats.unread), subtitle: "Notifikasi baru" },
        { title: "Hari Ini", value: String(stats.recentToday), subtitle: "Notifikasi hari ini" },
    ];

    // ... (JSX markup dibuang)
}
